  def _load_bm25_index(self, repo_id: int) -> bool:
      logging.info(f"📊 Loading BM25 index for repo_id={repo_id}")

      # Fast path: local disk (same Railway instance)
      try:
          bm25_path = self._get_bm25_path(repo_id)
          if os.path.exists(bm25_path):
              with open(bm25_path, 'rb') as f:
                  data = pickle.load(f)
              self.bm25 = data['bm25']
              self.chunk_ids = data['chunk_ids']
              logging.info(f"✅ Loaded BM25 from local disk ({len(self.chunk_ids):,} docs)")
              return True
      except Exception as e:
          logging.debug(f"Local BM25 not available: {e}")

      # Supabase Storage (persists across Railway restarts)
      try:
          from .supabase_client import get_supabase_client
          import gzip

          supabase = get_supabase_client()
          storage_path = self._get_bm25_storage_path(repo_id)

          logging.info(f"   Downloading BM25 from Storage: {storage_path}")
          response = supabase.storage.from_('repo-data').download(storage_path)

          if response:
              decompressed = gzip.decompress(response)
              data = pickle.loads(decompressed)
              self.bm25 = data['bm25']
              self.chunk_ids = data['chunk_ids']

              # Cache to local disk
              os.makedirs(os.path.dirname(self._get_bm25_path(repo_id)), exist_ok=True)
              with open(self._get_bm25_path(repo_id), 'wb') as f:

              logging.info(f"✅ Loaded BM25 from Supabase Storage ({len(self.chunk_ids):,} docs)")
              return True

      except Exception as e:
          logging.info(f"⚠️ BM25 not in Supabase Storage for repo_id={repo_id}: {e}")

      return False

          supabase = get_supabase_client()

          pickled = pickle.dumps(data)

          storage_path = self._get_bm25_storage_path(repo_id)

  #!/usr/bin/env python3
  """
  Save BM25/PageRank indexes for a repository already in the database.

  This is for repos that were uploaded before the Storage-based saving was implemented.

  Usage:
      python scripts/save_indexes_for_repo.py --repo-id 13

  Time: ~1-3 minutes for mega-repos (152K chunks)
  """

  import os
  import sys
  import asyncio
  import argparse
  import logging
  import time

  sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

  from dotenv import load_dotenv
  load_dotenv()

  logging.basicConfig(
      level=logging.INFO,
      format='%(asctime)s - %(levelname)s - %(message)s'
  )


  def save_indexes_for_repo(repo_id: int) -> bool:
      """
      Compute and save BM25/PageRank indexes for an existing repository.
      """
      from backend.api.data_storage import retrieve_chunks
      from backend.api.hybrid_retrieval import HybridRetriever, build_chunk_graph

      start_time = time.time()

      logging.info(f"🚀 SAVING INDEXES FOR REPO_ID={repo_id}")
      logging.info("=" * 60)

      # Step 1: Load chunks
      logging.info("📥 Step 1/4: Loading chunks from Supabase...")
      chunks = retrieve_chunks(repo_id)

      if not chunks:
          logging.error("❌ No chunks found for this repo_id!")
          return False

      logging.info(f"   ✅ Loaded {len(chunks):,} chunks")

      # Step 2: Build chunk graph
      logging.info("🔗 Step 2/4: Building chunk dependency graph...")
      graph_start = time.time()
      chunk_graph = build_chunk_graph(chunks)
      graph_time = time.time() - graph_start
      logging.info(f"   ✅ Built graph: {len(chunk_graph.nodes()):,} nodes, {len(chunk_graph.edges()):,} edges ({graph_time:.1f}s)")

      # Step 3: Build BM25 and PageRank
      logging.info("📊 Step 3/4: Building BM25 + computing PageRank...")
      logging.info("   (This takes 1-3 minutes for mega-repos)")

      build_start = time.time()
      hybrid_retriever = HybridRetriever(
          chunks=chunks,
          vector_store=None,  # Not needed for saving BM25/PageRank
          chunk_graph=chunk_graph,
          repo_id=repo_id,
          use_cached_indexes=False  # Force rebuild even if exists
      )
      build_time = time.time() - build_start

      logging.info(f"   ✅ BM25: {len(hybrid_retriever.chunk_ids):,} documents")
      logging.info(f"   ✅ PageRank: {len(hybrid_retriever.pagerank_scores):,} nodes")
      logging.info(f"   ✅ Build time: {build_time:.1f}s")

      # Step 4: Save to Supabase Storage
      logging.info("💾 Step 4/4: Saving to Supabase Storage...")
      save_start = time.time()
      success = hybrid_retriever.save_indexes(repo_id)
      save_time = time.time() - save_start

      if success:
          total_time = time.time() - start_time
          logging.info("=" * 60)
          logging.info(f"✅ SUCCESS: Saved indexes for repo_id={repo_id}")
          logging.info(f"   Total time: {total_time:.1f}s")
          logging.info(f"   - Graph build: {graph_time:.1f}s")
          logging.info(f"   - BM25/PageRank: {build_time:.1f}s")
          logging.info(f"   - Storage upload: {save_time:.1f}s")

          # Update preindexed_repos table
          try:
              from backend.api.supabase_client import get_supabase_client
              supabase = get_supabase_client()
              supabase.table('preindexed_repos').update({
                  'has_bm25_index': True,
                  'has_pagerank': True
              }).eq('repo_id', repo_id).execute()
              logging.info("   ✅ Updated preindexed_repos table")
          except Exception as e:
              logging.warning(f"   ⚠️ Could not update preindexed_repos: {e}")

          return True
      else:
          logging.error("❌ FAILED to save indexes")
          return False


  def main():
      parser = argparse.ArgumentParser(
          description='Save BM25/PageRank indexes for an existing repository'
      )
      parser.add_argument(
          '--repo-id',
          type=int,
          required=True,
          help='Repository ID (e.g., 13 for Kubernetes)'
      )
      args = parser.parse_args()

      success = save_indexes_for_repo(args.repo_id)
      sys.exit(0 if success else 1)


  if __name__ == '__main__':
      main()
